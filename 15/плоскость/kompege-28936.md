Полный бездумный перебор "плоскостных" задач зачастую очень тяжёл, но вот как вариант — перебор "десятичным" поиском.

```python
>>> def good(a):
...     for x in range(486):
...         for y in range(5*x+1):
...             if not ((x*y < a) or (5*x < y) or (486 <= x)):
...                 return False
...     return True
...
... def f(begin, end, step):
...     for a in range(begin, end, step):
...         if good(a):
...             print(a)
...             break
...
>>> f(0, 10**8, 10**7)
10000000
>>> f(0, 10**7, 10**6)
2000000
>>> f(10**6, 2*10**6, 10**5)
1200000
>>> f(11*10**5, 12*10**5, 10**4)
1180000
>>> f(117*10**4, 118*10**4, 10**3)
1177000
>>> f(1176*10**3, 1177*10**3, 10**2)
1176200
>>> f(11761*10**2, 11762*10**2, 10)
1176130
>>> f(1176120, 1176130, 1)
1176126
```
